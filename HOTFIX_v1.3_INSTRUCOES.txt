    def should_pause_globally(self) -> Tuple[bool, str]:
        """Determina se o sistema inteiro deve pausar por falta de atividade."""
        # Se 10+ ciclos globais vazios, sugere pausa longa
        if self.global_empty_cycles >= 10:
            return True, f"{self.global_empty_cycles} ciclos globais sem insercoes. Possivel bloqueio massivo."

        # Se todos os alvos conhecidos estao em backoff/exhausted
        if self.target_states:
            all_blocked = all(s.status in ("EXHAUSTED", "BLOCKED", "BACKOFF") for s in self.target_states.values())
            if all_blocked:
                return True, "Todos os alvos em backoff. Aguardando recuperacao."

        return False, ""

SUBSTITUA POR:
    def should_pause_globally(self) -> Tuple[bool, str]:
        """Determina se o sistema inteiro deve pausar por falta de atividade."""

        # [HOTFIX v1.3] ANTES de pausar, verifica se ha NOVOS itens na fila Supabase
        # que ainda nao foram processados (nao estao no target_states)
        try:
            import asyncio
            if hasattr(self.base_queue, 'db'):
                result = asyncio.get_event_loop().run_until_complete(
                    asyncio.to_thread(
                        self.base_queue.db.table("fila_coleta")
                        .select("id", count="exact")
                        .is_("locked_by", "null")
                        .not_.in_("status", ["CONCLUIDO", "SEM_DADOS_RECENTES"])
                        .execute
                    )
                )
                pending_count = getattr(result, "count", 0)
                if pending_count > len(self.target_states):
                    return False, f"{pending_count} itens pendentes na fila — continuando"
        except Exception:
            pass  # Fallback para comportamento anterior

        # Se 10+ ciclos globais vazios, sugere pausa longa
        if self.global_empty_cycles >= 10:
            return True, f"{self.global_empty_cycles} ciclos globais sem insercoes. Possivel bloqueio massivo."

        # Se todos os alvos conhecidos estao em backoff/exhausted
        if self.target_states:
            all_blocked = all(s.status in ("EXHAUSTED", "BLOCKED", "BACKOFF") for s in self.target_states.values())
            if all_blocked:
                return True, "Todos os alvos em backoff. Aguardando recuperacao."

        return False, ""

INSIRA DEPOIS:

            self.logger.warning(f"[HOTFIX v1.3] Todas as {health.sessions_blocked} sessoes bloqueadas. Tentando recuperacao...")
            try:
                if hasattr(session_pool, '_sessions'):
                    recovered = 0
                    for session in session_pool._sessions:
                        if hasattr(session, 'blocked_until') and session.blocked_until:
                            blocked_age = (datetime.now(timezone.utc) - session.blocked_until).total_seconds()
                            if blocked_age > 1800:  # Bloqueada ha mais de 30min
                                session.blocked_until = None
                                session.is_available = True
                                recovered += 1
                    if recovered > 0:
                        self.logger.info(f"[HOTFIX v1.3] {recovered} sessoes recuperadas.")
                        await self.ntfy.send(
                            title="Sentinela — Recuperacao",
                            message=f"{recovered} sessoes Instagram recuperadas apos cooldown.",
                            priority="default",
                            tags=["key"],
                        )
            except Exception as e:
                self.logger.error(f"[HOTFIX v1.3] Erro ao recuperar sessoes: {e}")

        """Retorna cor baseada no estado do agente."""
        if not self.agent.is_running:
            return "gray"
        if self.agent.consecutive_blocks >= 3:
            return "red"
        if self.agent.consecutive_blocks > 0:
            return "yellow"
        return "green"

SUBSTITUA POR:
    def _get_status_color(self) -> str:
        """Retorna cor baseada no estado do agente."""
        if not self.agent.is_running:
            return "gray"
        # [HOTFIX v1.3] Se fila vazia (0 alvos pendentes), mostra AZUL (IDLE) nao vermelho
        if hasattr(self.agent, '_smart_queue'):
            sq_stats = self.agent._smart_queue.get_stats()
            if sq_stats.get('targets_tracked', 0) == 0:
                return "blue"  # IDLE — nao ha trabalho, mas nao eh erro
        if self.agent.consecutive_blocks >= 3:
            return "red"
        if self.agent.consecutive_blocks > 0:
            return "yellow"
        return "green"
